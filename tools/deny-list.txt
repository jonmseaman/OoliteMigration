# Tier A / B symbol deny-list (docs/execution-model.md §7.5, docs/phases/0-safety-net.md 0.11).
#
# One extended-regular-expression per line; blank lines and #-comments ignored.
# A pattern that must match a literal '#' has to start with the bracket form '[#]', or the
# reader takes the whole line for a comment and the rule silently never runs.
# Consumed by tools/tier-a.sh, which greps a changed file for these and compares the hit
# count against the same file at the merge base — a *new* hit fails the tier, and by
# tools/guardrails.sh (bead oo-ptc), which applies the same baseline-relative rule to every
# changed source file in a branch rather than to one named TU.
#
# It has to be baseline-relative, not absolute: today the tree legitimately contains 3,828
# JS_* call sites across 102 files and links libgnustep-base everywhere. The regression
# this catches is reintroduction — a file that was migrated off SpiderMonkey or off
# GNUstep growing a call back. Once Phase 1 and Phase 2 land and the counts reach zero,
# baseline-relative and absolute mean the same thing, and nothing here has to change.

# --- SpiderMonkey (Phase 1, ADR-0002: replaced by quickjs-ng behind a JSEngine facade) ---
\bJS_[A-Za-z_][A-Za-z0-9_]*\b
\bJSRuntime\b
\bJSContext\b
\bjsval\b

# --- libgnustep-base (Phase 2: replaced by oofnd; the Objective-C runtime stays) ---
\bGNUSTEP_BASE_LIBRARY\b
[#]import[[:space:]]*<Foundation/
\bNSAutoreleasePool\b
[-]lgnustep-base\b
\bNS_DURING\b
\bNS_HANDLER\b
# The build-level spellings -lgnustep-base misses (bead oo-qps.2): an #include of a Foundation
# header; the library named as a quoted string (meson's dependency lists and
# find_library('...')); and as an item of a shell array assignment (tools/setup-windows.sh
# WINDEPS_PKGS). Deliberately NOT the bare word: comments in bridge headers and tools name the
# library in prose, and a new file's baseline is empty, so a bare-word pattern would fail
# legitimate work. The absolute "none left" check is tools/check-foundation-free.sh.
[#][[:space:]]*include[[:space:]]*<Foundation/
['"][g]nustep-base['"]
=\([^)]*\b[g]nustep-base\b

# Foundation's classes, protocols and functions by name (bead oo-qps.19, ADR-0055): the counts
# reached zero with oo-qps.18, so baseline-relative now means none. Enumerated, because ERE
# cannot say "NS* except the C types oofnd/objc/OOFoundationTypes.h keeps" (NSInteger, NSRange,
# NSPoint ... stay legal). Mac-fenced files keep theirs (a baseline, not an addition). The
# bridge headers deleted by oo-qps.16 / oo-snzn, and OOFoundationException.h, may not come back.
\b[N]S(String|Array|Dictionary|Set|Number|Data|Enumerator|Mutable[A-Za-z]+|Object|Log|C?(Parameter)?Assert[0-9]?|[A-Za-z]*Exception|UserDefaults|Bundle|FileManager|Date|Value|Null|CharacterSet|Scanner|Error|URL[A-Za-z]*|Task|Pipe|FileHandle|Thread|Lock|Condition|Timer|RunLoop|Notification[A-Za-z]*|Invocation|MethodSignature|Proxy|ProcessInfo|Zone|Copying|ComparisonResult|Ordered(Ascending|Same|Descending)|AllocateObject|ClassFromString|StringFromClass|SelectorFromString|StringFromSelector|MapTable|HashTable|SortDescriptor|AutoreleasePool)\b
[+]FoundationBridge\.h
"(OOStringBridge|OOFoundationBridge|OOPListView|OOFoundationException)\.h"

# --- SpiderMonkey build dependencies (Phase 1 item 1.7: deleted with the backend) ---
# NSPR ships with mozjs and only ever arrives as its transitive dependency; a PR_* call or an
# -lnspr on a link line means the old engine came back in through the build system.
\bmozillajs\b
\bmozjs\b
[-]lnspr[0-9]*\b
\bPR_(Malloc|Free|GetCurrentThread|Init|NewLock|Lock|Unlock)\b

# --- Warning suppression (CLAUDE.md rule 3) ---
# Kept here as well as in tools/guardrails.sh's suppression check because the two ask different
# questions: the deny-list is per-file and baseline-relative, so it also catches a suppression
# that arrives by COPYING an already-suppressed file (a copy has a fresh, empty baseline; a
# rename is compared against its old path and is correctly a no-op), which an added-lines scan
# can miss.
# That claim is only true if the two pattern sets actually agree. They diverged once: the
# _Pragma("...") operator was in neither, and _Pragma is the ONLY way to put a diagnostic
# suppression inside a macro — `#define QUIET_BEGIN _Pragma("GCC diagnostic push")` — so it is
# the natural spelling in a port. Keep this block and tools/guardrails.sh sup_patterns() in step.
# Bracket forms so this file does not match itself when it is the file being scanned.
[-]Wno-[A-Za-z0-9=_-]+
[#][[:space:]]*pragma[[:space:]]+(GCC|clang)[[:space:]]+diagnostic
[_]Pragma[[:space:]]*\([[:space:]]*"[^"]*diagnostic
__attribute__[[:space:]]*\(\([^)]*unused
